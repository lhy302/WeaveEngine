4.1 module_loader.py (M1-3) — 契约的 Python 侧机械层; 不 import
    module_registry/fastpath/engine; 策略层一律不做。
```python
class ModuleLoadError(Exception)
class ModuleValidationFailed(ModuleLoadError)   # 加载/校验失败 → 基座 catch 后弹出
class ModuleNotAvailable(ModuleLoadError)       # LOOM_NO_DLL 跳过/槽位空 → get 抛
class AssemblyError(ModuleLoadError)            # 官方组件坏/注册表结构错 → 不降级

LOOM_MODULE_MAGIC = 0x4D4F4F4C
LOOM_ABI_VERSION_CURRENT = (1 << 16) | 0
def loom_abi_make(major, minor); def loom_abi_major(v); def loom_abi_minor(v)
LOOM_DTYPE_F32,F64,I32,I64,U8 = 0,1,2,3,4

class LoomModuleInfo(Structure): magic u32, abi_version u32, module_id c_char*64,
    version c_char*16, official u32, provides_count u32, depends_count u32, reserved u32
class LoomTensor(Structure): data c_void_p, shape c_int64*4, ndim c_int32, dtype c_int32
def tensor_from_array(arr) -> (LoomTensor, keepalive)
    # np.ascontiguousarray; shape[0..ndim-1]=真实维度, 其余置 0; dtype 映射;
    # numpy 函数体内延迟 import; keepalive 调用方必须持有到调用结束

class ModuleHandle:   # 属性 path/module_id/version/official/abi_version/
                      #      provides/depends/inited
    @classmethod load(cls, path, expected_module_id=None)   # 失败抛 ModuleValidationFailed
    has(iface_name) -> bool          # 导出缺席 → False
    get_fn(name, restype, argtypes)  # dict 缓存; 已卸载抛 ModuleNotAvailable
    init() -> int                    # 已 inited 幂等返 0
    deinit() -> None                 # 未 inited 幂等
    unload() -> None                 # deinit+FreeLibrary+清缓存+移出 _LIVE
    hot_reload(new_path=None) -> ModuleHandle  # deinit→unload→load(同 module_id)→init

def no_dll_skips() -> set   # LOOM_NO_DLL: 未设→空集; '*'→{'*'}; 'a,b'→{'a','b'}
```
    load() 校验链 (每步失败抛 ModuleValidationFailed, 消息含路径+原因):
      ① 路径不存在 ② WinDLL 加载失败 ③ 六必需导出在位 (has 可选)
      ④ restype/argtypes 按契约 typedef ⑤ info 非 NULL / magic / ABI 主版本
      相等 / abi() 与 info 双口一致 ⑥ module_id·version 严格 utf-8 + 长度守约
      ⑦ provides/depends 枚举 (NULL/重复/超 4096/越界未返 NULL 皆失败)
      ⑧ expected_module_id 不符失败。
    其他: 模块级 _LIVE dict 持句柄防 GC (Nuitka 折叠防御); import 期
      assert sizeof(LoomModuleInfo)==104; FreeLibrary 失败仅 stderr warning。

4.2 module_registry.py (M1-2) — 注册表验证器 + 组装器 (策略层)
```python

    modules: dict[str, ModuleHandle]      # 已加载+已 init
    provides_map: dict[str, str]          # iface -> module_id

                                          #          writeback/conflict/init_failed
    registry_path: str
    def get(self, module_id) -> ModuleHandle       # 缺件抛 ModuleNotAvailable
    def pop_and_fallback(self, module_id, reason, persist=True) -> ModuleHandle

        # → 同槽位官方条目 load+validate+init → 返新句柄;
        # 弹的是官方 / 无官方可回退 → AssemblyError
    def deinit_all(self); def shutdown(self); def summary(self) -> str

def assemble(registry_path=None, base_dir=None, do_init=True, persist=True)
    -> AssemblyResult
    # registry_path 默认 <引擎根>/core_components/registry.json;
    # base_dir 默认引擎根; 条目 path 相对 base_dir 解析
```
    规则 (权威口径见 注册表格式与契约设计说明.txt §6/§7):
      · _format 不符/结构非法 → AssemblyError
      · 槽位=module_id; 候选序 = 用户注入(enabled) → 官方(enabled);
        同槽位同 officialness 重复条目 → AssemblyError
      · sha256 前置闸 (LoadLibrary 之前): official=false 且非空且不符 →
        不加载直接弹出; 官方条目 sha 不符 → AssemblyError
      · 加载/校验失败: 用户注入 → 弹出 + enabled=false 回写 + 官方顶回;
        官方 → AssemblyError
      · info.official 必须与条目 official 一致; provides/depends JSON 声明
        必须与 DLL 导出枚举逐集合一致 → 不符按失败处理
      · 接口双提供冲突 → 弹用户注入一方; 双官方 → AssemblyError
      · depends 未满足 → 依赖方弹出 (用户注入回退官方; 官方 → AssemblyError)
      · 依赖环 → AssemblyError (init 拓扑序无解)
      · init 按拓扑序; init 失败按弹出规则处理
      · LOOM_NO_DLL: '*' 全跳 / 指名槽位跳过 → report action=skipped,
        get 抛 ModuleNotAvailable
      · 回写用原子写 (临时文件 + os.replace), 只改 enabled 字段, 其余原样
      · 定点循环上限 (如 64 轮) 防死循环 → 超限 AssemblyError
