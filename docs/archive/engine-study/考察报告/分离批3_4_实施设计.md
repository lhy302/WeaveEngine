定案1【advance_logic 调 paintGL 层, 不动 _on_tick】
  现状链: ui/loom_gui.py:763 _on_tick (QTimer) → :780 e._update_camera()
  → :782 viewport.update() → :163 paintGL → :177-180 算 delta → e.render。
  _on_tick 一行不改 (它只管输入轮询+相机+调度重绘); advance_logic 插在
  paintGL 的 delta 计算与 e.render 之间。理由:
  a) delta 在 paintGL 现成 (:178), _on_tick 不算 dt;
  b) 逻辑帧必须与渲染帧严格同拍 (alpha 消费同一帧状态): 放 _on_tick 时
     窗口遮挡 Qt 跳过 paint 会逻辑快进、渲染不跑;
  c) QTimer 档位 60 (16ms) + 逻辑 30Hz = 稳定 2 步/帧 (accumulator 数学,
     与调用层无关), 30 档 = 1 步/帧, unlimited (1ms) = 0-1 步/帧。
  d) 改动面最小: GLViewport.paintGL 只加一行。

定案2【engine 侧 advance_logic 兜底 = 向后兼容总闸】
  render 直调方很多 (tests/*、build_entry._bench_scene:211/221、探针)。
  GLEngine.advance_logic 置 self._logic_advanced=True; render 开头兜底:
  is_playing 且未 advance 时自动调 advance_logic(delta_time), render 末尾
  清标志。⇒ 未接线的宿主行为不丢 (split 开: render 内跑 accumulator;
  split 关: 旧四行原样), 已接线的 paintGL/engine.run 不重复跑。

定案3【split 关 = 旧四行逐位保持; split 开才走 accumulator】
  LOOM_NO_SPLIT=1 或 设置 'logic_fps'<=0 → advance_logic 的 fallback 分支
  原样跑 旧 render is_playing 四行 (scripts/physics/audio/animations,
  dt=render_dt) —— A/B 基线逐位一致。物理 60Hz 子步 accumulator 只在
  split 开时启用 (见 3.3)。

--- 2.2 advance_logic / _clock_step / _split_enabled (新方法组) ---
    def _split_enabled(self):
        """分离开关: 设置 'logic_fps'>0 且非 LOOM_NO_SPLIT=1 且批1/2 文件可导入.
        每次进播放时求值 (toggle_play 不必重启). 测试桩 (__new__) 无字段 -> False."""
        if os.environ.get('LOOM_NO_SPLIT') == '1':
            return False
        try:
            lf = float(getattr(self, '_logic_fps_setting', 0) or 0)
        except Exception:
            return False
        return lf > 0

    def _ensure_split(self):
        """惰性建 clock/interp (进播放首帧调一次). 失败 -> _split_on=False 旧态."""
        if self._split_on:
            return
        try:
            from core_clock import FixedStepClock
            from interp_state import InterpState
        except Exception as e:
            print("[分离] core_clock/interp_state 不可用 (%r) -> 回退旧模式" % (e,))
            self._logic_fps_setting = 0
            return
        pf = float(getattr(self, '_physics_fps_setting', 60.0) or 60.0)
        self._clock = FixedStepClock(
            logic_fps=float(self._logic_fps_setting), physics_fps=pf)
        self._clock._on_logic_step = self._clock_step   # 逻辑步回调 (含物理子步)
        self._interp = InterpState()
        self._interp.invalidate()
        self._phys_acc = 0.0
        self._split_on = True
        print("[分离] 逻辑帧 %gHz / 物理 %gHz (maximum_logic_delta=%.2fs)"
              % (self._clock.logic_fps, pf, self._clock.maximum_logic_delta))

    def advance_logic(self, render_dt):
        """渲染帧开头调 (paintGL / engine.run). split 开: 跑 N 个逻辑步
        (脚本+物理子步+音频+动画, dt=恒定 fixed_dt) + 抓插值快照; split 关:
        原地跑旧四行一次 (dt=render_dt, 逐位旧行为). 幂等: 同一渲染帧
        render 内兜底不会重复 (见 3.1)."""
        self._logic_advanced = True
        self._interp_override = None
        if not self.is_playing:
            self._interp_gen += 1
            return 0                      # 编辑态: 零逻辑 (渲染照跑, alpha 恒 0)
        if not self._split_enabled():
            self._split_on = False
            self._logic_split_fallback(render_dt)     # 旧四行 (A/B 基线, 3.5)
            self._interp_gen += 1
            return 0
        self._ensure_split()
        if not self._split_on:
            self._logic_split_fallback(render_dt)
            self._interp_gen += 1
            return 0
        steps = self._clock.advance(min(render_dt, 0.1))
        self._interp.snap(self.models)    # 逻辑帧末抓 curr (prev 已滚, 批2 契约)
        self._interp_gen += 1
        self._logic_lps = (self._logic_lps * 0.9
                           + steps / max(render_dt, 1e-4) * 0.1)
        return steps

    def _clock_step(self, fixed_dt):
        """每个逻辑步 (clock.advance 内部回调, dt 恒定):
        脚本 -> 物理子步 -> 音频 -> 动画. 顺序=旧 render is_playing 四行
        (editor.py:1747-1750) 逐位同序, 仅 dt 换 fixed_dt."""
        self._profile('logic_step', lambda: self._clock_step_inner(fixed_dt))

    def _clock_step_inner(self, fixed_dt):
        self._run_scripts(fixed_dt)
        self._logic_physics(fixed_dt)     # 含 60Hz 物理子步 accumulator (3.3)
        self.update_audio(fixed_dt)
        self.update_animations(fixed_dt)

--- 3.2 逻辑步内物理语义 (红线: 脚本改 velocity 本逻辑帧生效) ---

    def _logic_physics(self, fixed_dt):
        """逻辑步内的物理: 独立 60Hz accumulator (报告 §7 坑1: 30Hz 单步 33ms
        Jolt 会穿透). 每逻辑步最多把物理推进到逻辑时钟位置, 不足部分留在
        _phys_acc (下一逻辑步补) —— 物理频率与逻辑频率解耦."""
        pf = float(getattr(self, '_physics_fps_setting', 60.0) or 60.0)
        pdt = 1.0 / pf
        self._phys_acc += fixed_dt
        n = 0
        while self._phys_acc >= pdt - 1e-9 and n < 8:   # 8 步上限=死亡螺旋保险
            self._profile('physics', lambda: self.update_physics(pdt, events=False))
            self._phys_acc -= pdt
            n += 1
        if n == 0 and self._phys_acc > 0:
            pass   # 物理时钟落后于逻辑: 本步不动 (30Hz 逻辑 + 60Hz 物理 = 隔步 2 次)
        if self._phys_acc > 0.25:                       # 积压丢弃 (maximum_logic_delta 精神)
            self._phys_acc = 0.0

    def _world_matrices_interp(self, models, ov):
        """插值路径世界矩阵: 与 compute_world_matrices 同算法 (S@R@T 批量 +
        层级批量 matmul), 但 pos/rot/scl 三平面取插值结果, 且**绕过**
        _fw_cache/_soa_last_keys 写入 (防静止缓存被插值态污染; 插值平面
        每渲染帧都变, 缓存无意义). 深度 DP/_batch_rotations(rot=) 全复用."""
        n = len(models)
        id2i = {id(m): i for i, m in enumerate(models)}
        parent = np.full(n, -1, dtype=np.int64)
        for i, m in enumerate(models):
            if m.parent is not None:
                parent[i] = id2i.get(id(m.parent), -1)
        depth = _fp_depths(parent)
        pos, rot, scl = ov
        S = np.zeros((n, 4, 4), dtype=np.float64)
        S[:, 0, 0], S[:, 1, 1], S[:, 2, 2] = scl[:, 0], scl[:, 1], scl[:, 2]
        S[:, 3, 3] = 1.0
        T = np.zeros((n, 4, 4), dtype=np.float64)
        T[:, 0, 0] = T[:, 1, 1] = T[:, 2, 2] = T[:, 3, 3] = 1.0
        T[:, 3, 0], T[:, 3, 1], T[:, 3, 2] = pos[:, 0], pos[:, 1], pos[:, 2]
        R = self._batch_rotations(models, rot=rot)
        local = S @ R @ T
        world = np.empty_like(local)
        for d in range(int(depth.max()) + 1):
            idx = np.where(depth == d)[0]
            if d == 0:
                world[idx] = local[idx]
            else:
                world[idx] = local[idx] @ world[parent[idx]]
        return world

六、批4 (P-A 物理线程化) 总览与定案
====================================================================
定案1【独立 threading.Thread, 不复用 FramePool】
  理由: a) 物理线程必须**独占**桥全部状态 (生产者-消费者, 无锁共享纪律),
  FramePool worker 是无状态任务池, 复用会让任意 worker 碰桥 (纪律破产);
  b) 常驻线程才能把 Jolt _w / _entries / _last_sync 的 CPU 缓存热住;
  c) ScriptRunner (parallel.py:251) 的"Event 对发+同步等待"模型正是本
  需要的骨架 —— 物理线程照抄该结构 (go/done Event + fn 提交), 但绑核用
  pool_mask (物理是重 C 段, 与渲染池抢 pool 核比抢脚本核安全)。
  复用=只复用 parallel.pin_current_thread 与 get_parallel 的 policy, 不用池。

定案2【搬移范围 = pre_step/step/post_step 三段整体, 一行不改物理逻辑】
  主线程每逻辑步: post 指令 (dt, gravity) → 不等 → 跑脚本/动画/音频 →
  同步点等待+回读。三段内部代码 (physics_bridge.py:826-1062) 原样搬进
  物理线程执行体 —— R1 红线"睡眠/唤醒/策略R/残速死带/静态升级销毁重建
  一行不改, 只是搬线程"。
定案3【指令 = 快照参数, 不共享可变对象】
  指令携带 (kind, dt, gravity, models_version); 模型列表/位形读取发生在
  物理线程内 (pre_step 本来就逐模型读属性) —— 主线程 post 后到同步点前
  **不碰任何模型属性** (脚本段在 post 之后跑, 天然满足; 动画改位形也
  排在等待之前, 见 6.2 时序)。
定案4【结构变更指令化】
  ensure_models 的差量输入 (self.models 列表) 由主线程在 post 指令时
  浅拷贝快照 (list(self.models), O(n) 500 球 µs 级), 物理线程拿快照跑
  ensure_models —— 避免"物理线程遍历时主线程改列表"的竞态。增删/重建
  (destroy_body/create_body) 全在物理线程内, 与 step 同线程, 无跨线程
  Jolt 调用。

class PhysicsPipeThread:
    """常驻物理线程 (生产者-消费者). 主线程 post(...) 提交一帧, wait() 同步
    等待+回读. 全部桥状态只物理线程触碰; 主线程只拿 post_step 之后的模型
    属性 (回读是物理线程写的, 同步点后主线程读 = happens-before 安全)."""
    def __init__(self, bridge, mask=0, timeout=1.0):
        import threading
        self._br = bridge
        self._mask = int(mask or 0)   # pool_mask (绑核; 0=不绑)
        self._timeout = float(timeout)
        self._lock = threading.Lock()
        self._go = threading.Event()
        self._done = threading.Event()
        self._job = None          # (models_snapshot, dt, gravity, events)
        self._err = None
        self._degraded = False    # 超时降级标志 (主线程直跑)
        self._t = threading.Thread(target=self._loop, name='loom-physics',
                                   daemon=True)
        self._t.start()

    def _loop(self):
        from parallel import pin_current_thread
        if self._mask:
            pin_current_thread(self._mask)
        while True:
            self._go.wait()
            self._go.clear()
            job = self._job
            if job is None:
                break                          # 停止哨兵
            models, dt, g, ev = job
            try:
                br = self._br
                br.ensure_models(models)
                br.pre_step(dt, g)
                br.step(dt)                    # C 层释放 GIL: 主线程 GIL 空闲
                br.post_step()
                self._err = None
            except BaseException as e:
                self._err = e
            finally:
                self._done.set()               # 无论成败唤醒主线程 (防卡死)

    def post(self, models, dt, gravity, events):
        """主线程: 提交本逻辑步物理任务 (非阻塞)."""
        with self._lock:
            self._job = (list(models), dt, gravity, events)
            self._done.clear()
            self._go.set()

    def wait(self):
        """同步点: 等物理帧完成. 超时 (>1s) = 报错+永久降级主线程直跑."""
        if not self._done.wait(self._timeout):
            self._degraded = True
            raise RuntimeError("物理线程同步超时 (%.1fs) -> 降级主线程直跑"
                               % self._timeout)
        if self._err is not None:
            e, self._err = self._err, None
            raise e                            # 桥异常沿用 engine 降级链

    def stop(self):
        """停止: 哨兵 None -> join. 幂等 (ScriptRunner.stop 同款)."""
        with self._lock:
            self._job = None
            self._go.set()
        self._t.join(timeout=2.0)
